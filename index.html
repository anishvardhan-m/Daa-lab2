1) Single Source Shortest Path

Dijkstra's Algorithm

Nonnegative weights; 0 means no edge. Vertices numbered 1 to n (maximum 19). Use small integer weights; finite distances must be less than 99999.

#include <stdio.h>
#define INF 99999
int main()
{
    int cost[20][20], dist[20], visited[20] = {0};
    int n, v, i, j, u, min, count;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if(n < 1 || n >= 20) return 1;
    printf("Enter cost matrix (0 for no edge):\n");
    for(i = 1; i <= n; i++)
        for(j = 1; j <= n; j++)
        {
            scanf("%d", &cost[i][j]);
            if(i != j && cost[i][j] == 0)
                cost[i][j] = INF;
        }
    printf("Enter source vertex: ");
    scanf("%d", &v);
    if(v < 1 || v > n) return 1;
    for(i = 1; i <= n; i++)
        dist[i] = cost[v][i];
    dist[v] = 0;
    visited[v] = 1;
    for(count = 1; count < n; count++)
    {
        min = INF;
        u = -1;
        for(i = 1; i <= n; i++)
            if(!visited[i] && dist[i] < min)
            {
                min = dist[i];
                u = i;
            }
        if(u == -1) break;
        visited[u] = 1;
        for(i = 1; i <= n; i++)
            if(!visited[i] && dist[u] + cost[u][i] < dist[i])
                dist[i] = dist[u] + cost[u][i];
    }
    printf("Shortest distances:\n");
    for(i = 1; i <= n; i++)
        if(i != v)
        {
            if(dist[i] == INF) printf("%d -> %d: unreachable\n", v, i);
            else printf("%d -> %d: %d\n", v, i, dist[i]);
        }
    return 0;
}

1 -> 3: 5

2) All Pairs Shortest Path

Floyd's Algorithm

Use 99999 for no edge and 0 on the diagonal. Maximum 19 vertices. Small integer weights; finite distances must be less than 99999. No negative cycles.

#include <stdio.h>
#define INF 99999
int main()
{
    int p[20][20], n, i, j, k;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if(n < 1 || n >= 20) return 1;
    printf("Enter cost matrix (99999 for no edge):\n");
    for(i = 1; i <= n; i++)
        for(j = 1; j <= n; j++)
            scanf("%d", &p[i][j]);
    for(k = 1; k <= n; k++)
        for(i = 1; i <= n; i++)
            for(j = 1; j <= n; j++)
                if(p[i][k] != INF && p[k][j] != INF &&
                   p[i][k] + p[k][j] < p[i][j])
                    p[i][j] = p[i][k] + p[k][j];
    printf("All-pairs shortest distances:\n");
    for(i = 1; i <= n; i++)
    {
        for(j = 1; j <= n; j++)
            if(p[i][j] == INF) printf("INF ");
            else printf("%d ", p[i][j]);
        printf("\n");
    }
    return 0;
}

3) Travelling Salesman

A) Dynamic Programming

Bitmask DP; O(n² × 2ⁿ) time. Complete cost matrix, 2–12 cities, costs 0–1000000000. Starts and ends at city 0.

#include <stdio.h>
#include <limits.h>
#define MAX 12
#define INF (LLONG_MAX / 4)
long long cost[MAX][MAX], dp[1 << MAX][MAX];
int n;

long long tsp(int mask, int pos)
{
    int city;
    long long ans = INF, value;
    if(mask == (1 << n) - 1)
        return cost[pos][0];
    if(dp[mask][pos] != -1)
        return dp[mask][pos];
    for(city = 0; city < n; city++)
        if(!(mask & (1 << city)))
        {
            value = cost[pos][city] + tsp(mask | (1 << city), city);
            if(value < ans) ans = value;
        }
    return dp[mask][pos] = ans;
}

int main()
{
    int i, j, mask;
    printf("Enter number of cities (2 to 12): ");
    scanf("%d", &n);
    if(n < 2 || n > MAX) return 1;
    printf("Enter complete cost matrix:\n");
    for(i = 0; i < n; i++)
        for(j = 0; j < n; j++)
        {
            scanf("%lld", &cost[i][j]);
            if(cost[i][j] < 0 || cost[i][j] > 1000000000) return 1;
        }
    for(mask = 0; mask < (1 << n); mask++)
        for(i = 0; i < n; i++)
            dp[mask][i] = -1;
    printf("Minimum tour cost = %lld\n", tsp(1, 0));
    return 0;
}

B) Brute Force

Enumerates every ordering of the remaining cities; O(n!) time. Complete cost matrix, 2–10 cities, costs 0–1000000000. Starts and ends at city 0.

#include <stdio.h>
#include <limits.h>
#define MAX 10
long long cost[MAX][MAX], best = LLONG_MAX;
int n, visited[MAX];

void tsp(int pos, int count, long long total)
{
    int city;
    if(count == n)
    {
        total += cost[pos][0];
        if(total < best) best = total;
        return;
    }
    for(city = 1; city < n; city++)
        if(!visited[city])
        {
            visited[city] = 1;
            tsp(city, count + 1, total + cost[pos][city]);
            visited[city] = 0;
        }
}

int main()
{
    int i, j;
    printf("Enter number of cities (2 to 10): ");
    scanf("%d", &n);
    if(n < 2 || n > MAX) return 1;
    printf("Enter complete cost matrix:\n");
    for(i = 0; i < n; i++)
        for(j = 0; j < n; j++)
        {
            scanf("%lld", &cost[i][j]);
            if(cost[i][j] < 0 || cost[i][j] > 1000000000) return 1;
        }
    visited[0] = 1;
    tsp(0, 1, 0);
    printf("Minimum tour cost = %lld\n", best);
    return 0;
}

4) N-Queens

Backtracking

Prints every valid arrangement. Q denotes a queen; * denotes an empty square. Use a small positive n (maximum 19).

#include <stdio.h>
#include <stdlib.h>
int x[20], count = 0;
int place(int k, int i)
{
    int j;
    for(j = 1; j < k; j++)
        if(x[j] == i || abs(x[j] - i) == abs(j - k))
            return 0;
    return 1;
}
void print_sol(int n)
{
    int i, j;
    printf("\nSolution %d:\n", ++count);
    for(i = 1; i <= n; i++)
    {
        for(j = 1; j <= n; j++)
            printf(x[i] == j ? "Q " : "* ");
        printf("\n");
    }
}
void queen(int k, int n)
{
    int i;
    for(i = 1; i <= n; i++)
        if(place(k, i))
        {
            x[k] = i;
            if(k == n) print_sol(n);
            else queen(k + 1, n);
        }
}
int main()
{
    int n;
    printf("Enter number of queens: ");
    scanf("%d", &n);
    if(n < 1 || n >= 20) return 1;
    queen(1, n);
    printf("\nTotal solutions = %d\n", count);
    return 0;
}

5) M-Colouring

Backtracking

Prints all valid colour assignments using colours 1 to m. Enter a symmetric 0/1 adjacency matrix for an undirected graph. Maximum 19 vertices and 19 colours; use small inputs.

#include <stdio.h>
#define MAX 20
int a[MAX][MAX], color[MAX], n, m, count = 0;

int safe(int vertex, int c)
{
    int i;
    if(a[vertex][vertex]) return 0;
    for(i = 1; i <= n; i++)
        if(a[vertex][i] && color[i] == c)
            return 0;
    return 1;
}

void coloring(int vertex)
{
    int c, i;
    if(vertex > n)
    {
        printf("Solution %d: ", ++count);
        for(i = 1; i <= n; i++)
            printf("%d ", color[i]);
        printf("\n");
        return;
    }
    for(c = 1; c <= m; c++)
        if(safe(vertex, c))
        {
            color[vertex] = c;
            coloring(vertex + 1);
            color[vertex] = 0;
        }
}

int main()
{
    int i, j;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if(n < 1 || n >= MAX) return 1;
    printf("Enter adjacency matrix of undirected graph:\n");
    for(i = 1; i <= n; i++)
        for(j = 1; j <= n; j++)
            scanf("%d", &a[i][j]);
    printf("Enter number of colours: ");
    scanf("%d", &m);
    if(m < 1 || m >= MAX) return 1;
    coloring(1);
    if(count == 0) printf("No colouring possible\n");
    printf("Total solutions = %d\n", count);
    return 0;
}

6) Hamiltonian Cycle

Backtracking

Prints one cycle starting and ending at vertex 1, visiting every vertex exactly once. Enter a symmetric 0/1 adjacency matrix for an undirected graph; 3–19 vertices.

#include <stdio.h>
#define MAX 20
int a[MAX][MAX], path[MAX], n;

int safe(int vertex, int pos)
{
    int i;
    if(!a[path[pos - 1]][vertex]) return 0;
    for(i = 1; i < pos; i++)
        if(path[i] == vertex) return 0;
    return 1;
}

int hamiltonian(int pos)
{
    int vertex;
    if(pos > n)
        return a[path[n]][path[1]] != 0;
    for(vertex = 2; vertex <= n; vertex++)
        if(safe(vertex, pos))
        {
            path[pos] = vertex;
            if(hamiltonian(pos + 1)) return 1;
            path[pos] = 0;
        }
    return 0;
}

int main()
{
    int i, j;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if(n < 3 || n >= MAX) return 1;
    printf("Enter adjacency matrix of undirected graph:\n");
    for(i = 1; i <= n; i++)
        for(j = 1; j <= n; j++)
            scanf("%d", &a[i][j]);
    path[1] = 1;
    if(hamiltonian(2))
    {
        printf("Hamiltonian cycle: ");
        for(i = 1; i <= n; i++)
            printf("%d ", path[i]);
        printf("%d\n", path[1]);
    }
    else
        printf("No Hamiltonian cycle\n");
    return 0;
}
